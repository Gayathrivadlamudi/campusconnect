3. A basic Storage Security Rule
match/b/{bucket}/o{}
??//the above line tells firebase to apply rules to storage at this location
match — matches a path.
{bucket} — wildcard representing the bucket name.
/o — represents the stored objects (files).

rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {
    match /profileImages/{userId}/{fileName} {
      allow read: if request.auth != null
                  && request.auth.uid == userId;
    }
  }
}

. Handle upload errors using try-catch
Future<void> uploadImage(File imageFile) async {
  try {
    final storageRef = FirebaseStorage.instance
        .ref()
        .child('profileImages')
        .child('user123')
        .child('profile.jpg');

    await storageRef.putFile(imageFile);

    print('Image uploaded successfully');
  } on FirebaseException catch (e) {
    print('Firebase error: ${e.code}');
    print('Message: ${e.message}');
  } catch (e) {
    print('Unexpected error: $e');
  }
}
